document.getElementById("csvBtn").onclick = () => { if (!himData || !herData) { alert("両方の回答データを読み込んでください"); return; } // ★ 質問リスト（質問番号＋質問内容） const questions = [ { no: "Q1", text: "朝起きて最初にする行動は何？" }, { no: "Q2", text: "朝の準備で毎日必ず通る場所はどこ？" }, { no: "Q3", text: "家の中で一番よく使う場所は？" }, { no: "Q4", text: "朝の動線で改善したい点はある？" }, // …あなたの62問をここに入れる（必要なら全部作るよ） ]; // ★ CSVのヘッダー const rows = [["質問番号","質問内容","さとし","ゆみこ"]]; // ★ 質問番号順に並べてCSV化 questions.forEach(q => { const no = q.no; const text = q.text; const him = himData[no] || ""; const her = herData[no] || ""; rows.push([no, text, him, her]); }); const csv = rows.map(r => r.join(",")).join("\n"); // ★★★ Excel文字化け防止：UTF-8 BOM を付与 ★★★ const BOM = new Uint8Array([0xEF, 0xBB, 0xBF]); const blob = new Blob([BOM, csv], { type: "text/csv" }); const a = document.createElement("a"); a.href = URL.createObjectURL(blob); a.download = "merged.csv"; a.click(); };
